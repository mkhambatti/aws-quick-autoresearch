SELECT * FROM (SELECT 'routes' AS kind, Month, Origin AS Origin, CAST(Dest AS VARCHAR) AS dim_s, COUNT(*) AS flights, COUNT(ArrDelay) AS observed_arrivals, SUM(ArrDelay) AS arrival_delay_minutes,

 SUM(CASE WHEN Cancelled = 1 THEN 1 ELSE 0 END) AS cancellations,
 SUM(CASE WHEN Diverted = 1 THEN 1 ELSE 0 END) AS diversions FROM data GROUP BY Month, Origin, Dest UNION ALL SELECT 'carriers' AS kind, Month, Origin AS Origin, CAST(UniqueCarrier AS VARCHAR) AS dim_s, COUNT(*) AS flights, COUNT(ArrDelay) AS observed_arrivals, SUM(ArrDelay) AS arrival_delay_minutes,
 SUM(CASE WHEN ArrDelay >= 15 THEN 1 ELSE 0 END) AS delayed_flights,

 SUM(CASE WHEN ArrDelay >= 15 THEN 1 ELSE 0 END) AS delayed_flights,

 SUM(CASE WHEN Diverted = 1 THEN 1 ELSE 0 END) AS diversions FROM data GROUP BY Month, Origin, FLOOR(CRSDepTime / 100.0) UNION ALL SELECT 'weekdays' AS kind, Month, Origin AS Origin, CAST(DayOfWeek AS VARCHAR) AS dim_s, COUNT(*) AS flights, COUNT(ArrDelay) AS observed_arrivals, SUM(ArrDelay) AS arrival_delay_minutes,
